# pkg.install developer flow, driven over the operator console.
# Anchors: console-service drops bytes outside an armed readline, and the
# setup-wizard owns the stream first on factory-fresh volumes (its own
# deadline applies defaults after ~3 idle steps). The prompt glyph is only
# rendered on typing, never flushed standalone, so anchoring uses milestone
# text instead of PROMPT_PATTERN.
expect: serviceos shell ready
expect: setup complete
send: status health
expect: system health @tick
# quiesce: last rendered status row proves the render finished and the
# operator readline is re-armed before the next line is typed.
expect: attention: problems=
# resend guards against console readline re-arm races that can silently
# swallow a whole typed line between commands (WP3 additive convention).
resend: pkg install developer until installed developer-service|install failed:
